import logging from telegram import InlineKeyboardButton, InlineKeyboardMarkup, Update from telegram.ext import Application, CallbackQueryHandler, CommandHandler, ContextTypes # Configurazione Logging logging.basicConfig(level=logging.INFO) # ========================================== # CONFIGURAZIONE E VARIABILI DI STATO # ========================================== # Imposta su True per attivare il Trailing SL, False per disattivarlo USA_TRAILING_SL = True # Stato fittizio della posizione (giusto per l'esempio) posizione_attiva = { "open": True, "tipo": "LONG", "entry_price": 25000.0, "current_price": 25300.0, "sl": 24800.0, "tp": 26000.0 } # ========================================== # LOGICA DI TRADING (TRAILING STOP LOSS) # ========================================== def gestisci_trailing_stop(): """Gestisce il movimento dello stop loss se l'opzione è attiva.""" if not USA_TRAILING_SL or not posizione_attiva["open"]: return # Esempio matematico per un'operazione LONG: # Se il prezzo sale, lo stop loss lo segue a una distanza fissa distanza_sl = 200.0 # Distanza fissa dal prezzo corrente if posizione_attiva["tipo"] == "LONG": nuovo_sl = posizione_attiva["current_price"] - distanza_sl # Lo stop loss sale solo, non scende mai if nuovo_sl > posizione_attiva["sl"]: posizione_attiva["sl"] = nuovo_sl logging.info(f"Trailing SL aggiornato a: {posizione_attiva['sl']}") # ========================================== # INTERFACCIA GRAFICA E PULSANTI # ========================================== async def mostra_grafico_e_comandi(update: Update, context: ContextTypes.DEFAULT_TYPE): """Invia il grafico con i tasti di chiusura immediata posizionati subito sotto.""" # Stato del Trailing SL da mostrare visivamente sul report stato_trailing = "🟢 ATTIVO" if USA_TRAILING_SL else "🔴 DISATTIVATO" testo_report = ( f"📊 *Grafico e Stato Posizione*\n" f"🔹 Tipo: {posizione_attiva['tipo']}\n" f"🔹 Prezzo Ingresso: {posizione_attiva['entry_price']}\n" f"🔹 Prezzo Corrente: {posizione_attiva['current_price']}\n" f"🔹 Trailing SL: {stato_trailing}\n" f"🛑 SL: {posizione_attiva['sl']} | 🎯 TP: {posizione_attiva['tp']}" ) # Tasti posizionati vicini al grafico per la chiusura manuale rapida tastiera = [ [ InlineKeyboardButton("❌ CHIUDI POSIZIONE", callback_data="chiudi_totale") ], [ InlineKeyboardButton("⚠️ Chiudi 50%", callback_data="chiudi_50"), InlineKeyboardButton("🔄 Aggiorna", callback_data="aggiorna_grafico") ] ] reply_markup = InlineKeyboardMarkup(tastiera) # Invia l'immagine del grafico con la tastiera agganciata sotto di essa # Sostituisci 'url_o_path_grafico' con il file effettivo del tuo grafico url_o_path_grafico = "https://via.placeholder.com/600x400.png?text=Grafico+Trading" await context.bot.send_photo( chat_id=update.effective_chat.id, photo=url_o_path_grafico, caption=testo_report, reply_markup=reply_markup, parse_mode="Markdown" ) # ========================================== # GESTIONE DELLE AZIONI (CALLBACK) # ========================================== async def gestisci_pulsanti(update: Update, context: ContextTypes.DEFAULT_TYPE): """Gestisce la chiusura manuale quando l'utente clicca sui tasti sotto il grafico.""" query = update.callback_query await query.answer() if query.data == "chiudi_totale": posizione_attiva["open"] = False await query.edit_message_caption(caption="❌ **Posizione chiusa manualmente con successo.**") elif query.data == "chiudi_50": await query.edit_message_caption(caption="⚠️ **Chiuso il 50% della posizione. Restante in corsa.**") elif query.data == "aggiorna_grafico": # Simula un cambio prezzo e aggiorna il trailing se attivo posizione_attiva["current_price"] += 100 gestisci_trailing_stop() # Rigenera la schermata aggiornata stato_trailing = "🟢 ATTIVO" if USA_TRAILING_SL else "🔴 DISATTIVATO" nuovo_testo = ( f"📊 *Grafico Aggiornato*\n" f"🔹 Prezzo Corrente: {posizione_attiva['current_price']}\n" f"🔹 Trailing SL: {stato_trailing}\n" f"🛑 SL: {posizione_attiva['sl']} | 🎯 TP: {posizione_attiva['tp']}" ) await query.edit_message_caption(caption=nuovo_testo, reply_markup=query.message.reply_markup, parse_mode="Markdown") # ========================================== # MAIN APPLICATION # ========================================== def main(): # Inserisci qui il token del tuo bot Telegram TOKEN = "IL_TUO_TELEGRAM_BOT_TOKEN" app = Application.builder().token(TOKEN).build() # Comandi app.add_handler(CommandHandler("trade", mostra_grafico_e_comandi)) # Callback dei bottoni sotto il grafico app.add_handler(CallbackQueryHandler(gestisci_pulsanti)) print("Bot in esecuzione...") app.run_polling() if __name__ == "__main__": main()
